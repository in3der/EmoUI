# 전체 스택 동작 확인
python -c "
import tensorflow as tf, onnx, onnxruntime
from hsemotion_onnx.facial_emotions import HSEmotionRecognizer
import numpy as np
r = HSEmotionRecognizer(model_name='enet_b0_8_va_mtl')
_, scores = r.predict_emotions(np.zeros((224,224,3), dtype=np.uint8), logits=False)
print('TF:', tf.__version__, '| onnx:', onnx.__version__, '| ort:', onnxruntime.__version__)
print('valence:', round(float(scores[8]),4), '| arousal:', round(float(scores[9]),4))
print('✅ 전체 스택 정상')
"
